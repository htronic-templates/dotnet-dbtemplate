Docker Compose Image

Example docker compose file compose.yml

services:
sqlserver:
image: mcr.microsoft.com/mssql/server:2025-latest
container_name: sqlserver1
environment:
ACCEPT_EULA: "Y"
MSSQL_PID: "Developer"
MSSQL_SA_PASSWORD: "%DemoPassword1234"
ports:
- "1433:1433"
volumes:
- sqlserverdata:/var/opt/mssql
